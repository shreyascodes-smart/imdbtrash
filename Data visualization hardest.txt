import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import warnings

warnings.filterwarnings("ignore")

sns.set_style("whitegrid")
plt.rcParams["figure.dpi"] = 100

url = "https://raw.githubusercontent.com/curran/data/gh-pages/superstoreSales/superstoreSales.csv"
df = pd.read_csv(url, encoding="latin1")

print("Shape:", df.shape)
print(df.head())

def clean_currency(x):
    if isinstance(x, str):
        x = x.replace("$", "").replace(",", "").strip()

        if x.startswith("(") and x.endswith(")"):
            x = "-" + x[1:-1]

        try:
            return float(x)
        except ValueError:
            return np.nan

    return x

currency_cols = [
    "Sales", "Profit", "Unit Price",
    "Shipping Cost", "Product Base Margin"
]

for col in currency_cols:
    df[col] = df[col].apply(clean_currency)

df["Order Date"] = pd.to_datetime(df["Order Date"])
df["Ship Date"] = pd.to_datetime(df["Ship Date"])
df["Order Month"] = (
    df["Order Date"].dt.to_period("M").dt.to_timestamp()
)
df["Order Year"] = df["Order Date"].dt.year

print(df.dtypes)
print(df.isnull().sum())

monthly_sales = df.groupby("Order Month")["Sales"].sum().reset_index()

plt.figure(figsize=(12, 5))
sns.lineplot(
    data=monthly_sales,
    x="Order Month",
    y="Sales",
    marker="o",
    color="steelblue"
)
plt.title("Monthly Sales Trend", fontsize=14, fontweight="bold")
plt.xlabel("Month")
plt.ylabel("Total Sales ($)")
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

cat_sales = (
    df.groupby("Product Category")["Sales"]
    .sum().sort_values(ascending=False).reset_index()
)

plt.figure(figsize=(7, 5))
sns.barplot(
    data=cat_sales,
    x="Product Category",
    y="Sales",
    palette="viridis"
)
plt.title("Total Sales by Product Category", fontsize=14, fontweight="bold")
plt.xlabel("Product Category")
plt.ylabel("Total Sales ($)")
plt.tight_layout()
plt.show()

cat_profit = (
    df.groupby("Product Category")["Profit"]
    .sum().sort_values(ascending=False).reset_index()
)

plt.figure(figsize=(7, 5))
colors = [
    "seagreen" if p > 0 else "firebrick"
    for p in cat_profit["Profit"]
]

sns.barplot(
    data=cat_profit,
    x="Product Category",
    y="Profit",
    palette=colors
)
plt.title("Total Profit by Product Category", fontsize=14, fontweight="bold")
plt.xlabel("Product Category")
plt.ylabel("Total Profit ($)")
plt.axhline(0, color="black", linewidth=0.8)
plt.tight_layout()
plt.show()

region_sales = (
    df.groupby("Region")["Sales"].sum().sort_values(ascending=False)
)

plt.figure(figsize=(7, 7))
plt.pie(
    region_sales,
    labels=region_sales.index,
    autopct="%1.1f%%",
    startangle=90,
    colors=sns.color_palette("pastel")
)
plt.title("Sales Share by Region", fontsize=14, fontweight="bold")
plt.tight_layout()
plt.show()

plt.figure(figsize=(8, 5))
sns.boxplot(
    data=df,
    x="Customer Segment",
    y="Sales",
    palette="Set2",
    showfliers=False
)
plt.title("Sales Distribution by Customer Segment", fontsize=14, fontweight="bold")
plt.xlabel("Customer Segment")
plt.ylabel("Sales ($)")
plt.tight_layout()
plt.show()

top_subcat = (
    df.groupby("Product Sub-Category")["Sales"]
    .sum().sort_values(ascending=False).head(10)
)

plt.figure(figsize=(9, 6))
sns.barplot(
    x=top_subcat.values,
    y=top_subcat.index,
    palette="mako"
)
plt.title("Top 10 Product Sub-Categories by Sales", fontsize=14, fontweight="bold")
plt.xlabel("Total Sales ($)")
plt.ylabel("Sub-Category")
plt.tight_layout()
plt.show()

plt.figure(figsize=(8, 5))
sns.scatterplot(
    data=df,
    x="Discount",
    y="Profit",
    hue="Product Category",
    alpha=0.6
)
plt.title("Discount vs Profit", fontsize=14, fontweight="bold")
plt.xlabel("Discount")
plt.ylabel("Profit ($)")
plt.axhline(0, color="black", linewidth=0.8, linestyle="--")
plt.tight_layout()
plt.show()

numeric_df = df[
    ["Sales", "Profit", "Discount", "Order Quantity",
     "Unit Price", "Shipping Cost"]
]

plt.figure(figsize=(8, 6))
sns.heatmap(
    numeric_df.corr(),
    annot=True,
    cmap="coolwarm",
    fmt=".2f",
    linewidths=0.5
)
plt.title("Correlation Heatmap of Numeric Features", fontsize=14, fontweight="bold")
plt.tight_layout()
plt.show()

plt.figure(figsize=(7, 5))
sns.countplot(
    data=df,
    x="Ship Mode",
    order=df["Ship Mode"].value_counts().index,
    palette="crest"
)
plt.title("Number of Orders by Ship Mode", fontsize=14, fontweight="bold")
plt.xlabel("Ship Mode")
plt.ylabel("Order Count")
plt.tight_layout()
plt.show()
